<script>

const API_URL = "https://script.google.com/macros/s/AKfycbz47LCd1aYpTO0gteNXnlzYVZ4hBk04grk6ezj0Xe6xclm9PhGgMc27qKDHTP3mMVvg/exec";

const params = new URLSearchParams(window.location.search);
const materialId = params.get("material") || "N5-2012";

fetch(`${API_URL}?action=materials`)
.then(res => res.json())
.then(materials => {

    const material = materials.find(m => m.materialId === materialId);

    if (material) {
        document.getElementById("examTitle").textContent = material.name;
    }

});

const answerArea = document.getElementById("answerArea");

// 回答保存用
let answers = {};

// 保存キー（教材ごと）
const storageKey = "answers_" + materialId;

// 保存データ読み込み
const saved = localStorage.getItem(storageKey);

if (saved) {
    answers = JSON.parse(saved);
}

fetch(`${API_URL}?action=answers&materialId=${materialId}`)
    .then(res => res.json())

.then(data => {

    let html = "";

    // セクションごとにまとめる
    const sections = {};

    data.forEach(item => {

        const section = item.type;

        if (!sections[section]) {
            sections[section] = [];
        }

        sections[section].push(item);

    });

    // HTML生成
    Object.keys(sections).forEach(section => {

        let sectionTitle = section;

        if (section === "goi") sectionTitle = "語彙（goi）";
        if (section === "bunpou") sectionTitle = "文法（bunpou）";
        if (section === "dokkai") sectionTitle = "読解（dokkai）";

        if (section.startsWith("choukai")) {
            const no = section.replace("choukai", "");
            sectionTitle = `聴解${no}（${section}）`;
        }

        html += `<div class="sectionTitle">${sectionTitle}</div>`;

        sections[section].forEach((item, index) => {

            const answerKey =
                section === "dokkai"
                    ? section + "-" + item.number
                    : section + "-" + (index + 1);

            const displayNumber =
                section === "dokkai"
                    ? item.number
                    : (index + 1);

            html += `
            <div class="answerRow" data-key="${answerKey}">

                <div class="answerNumber" data-key="${answerKey}">
                    問題${displayNumber}
                </div>

                <div class="answerButtons">
                    <button>1</button>
                    <button>2</button>
                    <button>3</button>
                    <button>4</button>
                </div>

            </div>
            `;
        });

    });

    answerArea.innerHTML = html;

    // 保存された回答を復元
    document.querySelectorAll(".answerButtons").forEach((group, index) => {

        const answerKey = group.closest(".answerRow").dataset.key;
        const savedAnswer = answers[answerKey];

        if (!savedAnswer) return;

        const buttons = group.querySelectorAll("button");

        buttons.forEach(button => {

            if (button.textContent === savedAnswer) {
                button.classList.add("selected");
            }

        });

    });

    // ボタン処理
        document.querySelectorAll(".answerButtons").forEach((group,index)=>{

            const buttons = group.querySelectorAll("button");

            buttons.forEach(button=>{

                button.onclick=()=>{

                    buttons.forEach(b=>b.classList.remove("selected"));

                    button.classList.add("selected");

                    // 回答保存
                    const answerKey = group.closest(".answerRow").dataset.key;

                    answers[answerKey] = button.textContent;

                    localStorage.setItem(
                        storageKey,
                        JSON.stringify(answers)
                    );

                    console.log("保存しました", answers);

                };

            });

        });

        });

       document.getElementById("scoreButton").onclick = () => {

       document.getElementById("loadingOverlay").style.display = "flex";

       const button = document.getElementById("scoreButton");
       button.disabled = true;
       button.textContent = "採点中...";

           fetch(API_URL, {
               method: "POST",
               contentType: "application/json",
               body: JSON.stringify({
                   userName: userName.value.trim(),
                   materialId: materialId,
                   answers: answers
               })
           })

           .then(res => res.json())
           .then(result => {

               console.log(result);

               if (result.success) {

               document.getElementById("loadingOverlay").style.display = "none";

               const button = document.getElementById("scoreButton");
               button.disabled = false;
               button.textContent = "採点する";

               document.querySelectorAll(".answerNumber").forEach(el => {
                   el.dataset.result = "";
               });

               Object.keys(result.questionResult).forEach(question => {

                   const el = document.querySelector(
                       `.answerNumber[data-key="${question}"]`
                   );

                   if (!el) return;

                   if (result.questionResult[question]) {

                       el.innerHTML += ` <span class="resultCorrect">⭕</span>`;

                   } else {

                       el.innerHTML += `
                           <span class="resultWrong">❌</span>
                           <span class="correctAnswer"> 正解：${result.correctAnswers[question]}</span>
                       `;

                   }

               });

                   document.getElementById("resultArea").innerHTML = `
                   <h2>採点結果</h2>

                   <div>総合　${result.score} / ${result.total}</div>

                   <div>語彙　${result.sectionScore.goi.score} / ${result.sectionScore.goi.total}</div>

                   <div>文法　${result.sectionScore.bunpou.score} / ${result.sectionScore.bunpou.total}</div>

                   <div>読解　${result.sectionScore.dokkai.score} / ${result.sectionScore.dokkai.total}</div>

                   <div>聴解　${result.sectionScore.choukai.score} / ${result.sectionScore.choukai.total}</div>
                   `;

                   localStorage.removeItem(storageKey);


                   location.href = "#top";

               }

           })

           .catch(err => {

               console.error(err);

           });

       };

               const startButton = document.getElementById("startButton");
               const startOverlay = document.getElementById("startOverlay");
               const userName = document.getElementById("userName");

               const savedName = localStorage.getItem("userName");

               if (savedName) {
                   userName.value = savedName;
               }

               startButton.addEventListener("click", () => {
                   // ...
               });

               startButton.addEventListener("click", () => {

                   if (userName.value.trim() === "") {
                       alert("Please enter your name.");
                       userName.focus();
                       return;
                   }

                   localStorage.setItem("userName", userName.value.trim());

                   startOverlay.style.display = "none";

               });


</script>
