<script>

let dataA = [];
let dataB = [];
let index = 0;

// URLパラメータ（?lesson=21）
const lesson =
  new URLSearchParams(location.search).get("lesson") || 1;

// ===== テスト開始前の入口画面 =====

const startScreen =
  document.getElementById("startScreen");

const startLessonTitle =
  document.getElementById("startLessonTitle");

const startIssueId =
  document.getElementById("startIssueId");

const startTestButton =
  document.getElementById("startTestButton");

startLessonTitle.textContent =
  "Lesson " + lesson;

startTestButton.addEventListener("click", () => {

  startScreen.style.display = "none";

});

window.addEventListener("DOMContentLoaded", () => {
  document.getElementById("lessonTitle").textContent =
    "Lesson " + lesson;
});

// Apps Script API
const API_URL =
  "https://script.google.com/macros/s/AKfycbyaAxNzUzCZCQmwpcWpMantJdlzlUDFPLzh3QKOgOdoDv03ls1kNoxA81aSYeZ1PIgsLQ/exec";

function makeImageUrl(id){
  return "https://drive.google.com/thumbnail?id=" + id + "&sz=w1000";
}

function draw(){

  // データがまだ読み込まれていない場合は何もしない
  if(dataA.length === 0 || dataB.length === 0){
    return;
  }

  if(index >= dataA.length){
    index = 0;
  }

  const urlA = makeImageUrl(dataA[index].id);
  const urlB = makeImageUrl(dataB[index].id);

  document.getElementById("A").innerHTML =
    '<div class="card"><img src="'+urlA+'"></div>' +
    '<div class="label">A.' + (index+1) + '</div>';

  document.getElementById("B").innerHTML =
    '<div class="card"><img src="'+urlB+'"></div>' +
    '<div class="label">B.' + (index+1) + '</div>';

}

function next(){
  index++;
  draw();
}

fetch(API_URL + "?lesson=" + lesson + "&mode=json")
  .then(response => response.json())
  .then(data => {

      dataA = data.A;
      dataB = data.B;

      startIssueId.textContent =
        "発行番号：" + data.issueId;

      window.currentIssueId = data.issueId;

      draw();

    setInterval(next, 20000);

  })
  .catch(error => {
    console.error(error);
  });

document.getElementById("previewAQuestion")
  .addEventListener("click", () => {
    showPreview("A", false);
  });

document.getElementById("previewAAnswer")
  .addEventListener("click", () => {
    showPreview("A", true);
  });

document.getElementById("previewBQuestion")
  .addEventListener("click", () => {
    showPreview("B", false);
  });

document.getElementById("previewBAnswer")
  .addEventListener("click", () => {
    showPreview("B", true);

document.getElementById("previewClose")
  .addEventListener("click", () => {
    document.getElementById("previewModal")
      .classList.remove("show");
  });

document.getElementById("previewSavePng")
  .addEventListener("click", savePreviewAsPng);

async function savePreviewAsPng(){

  const modal = document.getElementById("previewModal");

  const clone = modal.cloneNode(true);

  clone.classList.add("show");

  clone.style.position = "fixed";
  clone.style.left = "0";
  clone.style.top = "0";
  clone.style.transform = "none";
  clone.style.width = "794px";
  clone.style.height = "1123px";
  clone.style.maxWidth = "none";
  clone.style.maxHeight = "none";
  clone.style.overflow = "hidden";
  clone.style.padding = "38px";
  clone.style.boxSizing = "border-box";

  const closeButton = clone.querySelector("#previewClose");
  const saveButton = clone.querySelector("#previewSavePng");

  if(closeButton){
    closeButton.style.display = "none";
  }

  if(saveButton){
    saveButton.style.display = "none";
  }

  document.body.appendChild(clone);

  try{

    const canvas = await html2canvas(clone, {
      backgroundColor: "#ffffff",
      scale: 3,
      width: 794,
      height: 1123,
      windowWidth: 794,
      windowHeight: 1123,
      scrollX: 0,
      scrollY: 0,
      useCORS: true
    });

    const link = document.createElement("a");

    link.download =
      "画像" +
      document.getElementById("previewTitle").textContent +
      "_" +
      (window.currentIssueId || "") +
      ".png";

    link.href = canvas.toDataURL("image/png");

    link.click();

  } finally {

    clone.remove();

  }
}

function showPreview(type, isAnswer){

  const list =
    type === "A" ? dataA : dataB;

  document.getElementById("previewTitle").textContent =
    "画像" + type +
    (isAnswer ? "　答えプレビュー" : "　問題プレビュー");

  document.getElementById("previewIssue").textContent =
    "発行番号：" + (window.currentIssueId || "");

  let leftHTML = "";
  let rightHTML = "";

  for(let i = 0; i < 20; i++){

    const item = list[i];

    if(!item) continue;

    const answer =
      isAnswer
        ? '<span class="preview-answer">' +
          item.reading +
          '</span>'
        : "　　　　";

    const html =
      '<div class="preview-row">' +
      (i + 1) +
      ". " +
      item.kanji +
      "（" +
      answer +
      "）" +
      "</div>";

    if(i < 10){
      leftHTML += html;
    }else{
      rightHTML += html;
    }
  }

  document.getElementById("previewLeft").innerHTML =
    leftHTML;

  document.getElementById("previewRight").innerHTML =
    rightHTML;

  document.getElementById("previewModal")
    .classList.add("show");
}
