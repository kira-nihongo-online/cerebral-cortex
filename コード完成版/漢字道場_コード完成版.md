<script>
  let cards = [];
  let currentIndex = 0;
  let showingFront = true;

  /* ★ 追加：スコア管理 */
  let correctCount = 0;
  let answered = [];

  const API_URL = "https://script.google.com/macros/s/AKfycbz7D_uzRRD9ARfOpK_MQ4IZ1YifyxqJmNPIRFijQJo4zSu14gbI9VnmcOzIwVMRi88/exec";

  const lesson = new URLSearchParams(location.search).get("lesson");

  const img1 = document.getElementById("img1");

  fetch(API_URL + "?lesson=" + lesson)
    .then(response => response.json())
    .then(function(data) {

      cards = data || [];
      currentIndex = 0;
      showingFront = true;

      answered = new Array(cards.length).fill(false);
      correctCount = 0;
      updateScore();

      if (cards.length > 0) {
        displayCard();
        preloadNext();
        updateScore();
      }
    });

  function displayCard() {
    if (cards.length === 0) return;

    img1.src = showingFront
      ? cards[currentIndex].front
      : cards[currentIndex].back;
  }

  function preloadNext() {
    if (cards.length === 0) return;

    const nextIndex = (currentIndex + 1) % cards.length;

    new Image().src = cards[currentIndex].back;
    new Image().src = cards[nextIndex].front;
  }

  function nextCard() {

    if (cards.length === 0) return;

    if (showingFront) {
      showingFront = false;
    } else {
      showingFront = true;
      currentIndex++;
      if (currentIndex >= cards.length) currentIndex = 0;
    }

    displayCard();
    preloadNext();
    updateScore();
  }


  function previousCard() {

    if (cards.length === 0) return;

    if (!showingFront) {

      // 裏 → 表
      showingFront = true;

    } else {

      // 表 → 前のカードの裏
      currentIndex--;
      if (currentIndex < 0) {
        currentIndex = cards.length - 1;
      }

      showingFront = false;
    }

    displayCard();
    preloadNext();
    updateScore();
  }


  document.addEventListener("click", function(e) {
    if (e.target.closest(".answer-area")) return;
    nextCard();
  });


  document.addEventListener("keydown", function(e) {

    if (e.key === "Enter") {
      e.preventDefault();
      nextCard();
    }

    if (e.key === "Backspace") {
      e.preventDefault();
      previousCard();
    }

  });
  /* ★ 追加：スコア表示更新 */
  function updateScore() {

    document.getElementById("scoreArea").textContent =
      "正解：" + correctCount + " / " + cards.length;

    document.getElementById("progressArea").textContent =
      (currentIndex + 1) + " / " + cards.length;
  }

  /* ★ 追加：正解ボタン動作 */
  document.getElementById("correctBtn").addEventListener("click", function(e) {
    e.stopPropagation();

    if (!answered[currentIndex]) {
      answered[currentIndex] = true;
      correctCount++;
      updateScore();
    }
  });

  // ===============================
  // クリアボタン
  // ===============================
  document.getElementById("clearBtn").addEventListener("click", function(e) {

      e.stopPropagation();

      ctx.clearRect(0, 0, canvas.width, canvas.height);

  });

  // ===============================
  // Canvas 初期設定
  // ===============================
  const canvas = document.getElementById("writeArea");
  const ctx = canvas.getContext("2d");

  function resizeCanvas() {

      const rect = canvas.getBoundingClientRect();

      canvas.width = rect.width;
      canvas.height = rect.height;

      ctx.lineWidth = 3;
      ctx.lineCap = "round";
      ctx.strokeStyle = "#000000";
  }

  resizeCanvas();

  window.addEventListener("resize", resizeCanvas);

  // ===============================
  // 描画処理
  // ===============================
  let drawing = false;

  function getPos(e) {

      const rect = canvas.getBoundingClientRect();

      const touch = e.touches ? e.touches[0] : e;

      return {
          x: touch.clientX - rect.left,
          y: touch.clientY - rect.top
      };
  }

  function startDraw(e) {

      drawing = true;

      const p = getPos(e);

      ctx.beginPath();
      ctx.moveTo(p.x, p.y);

      e.preventDefault();
  }

  function draw(e) {

      if (!drawing) return;

      const p = getPos(e);

      ctx.lineTo(p.x, p.y);
      ctx.stroke();

      e.preventDefault();
  }

  function endDraw() {

      drawing = false;
  }

  // ===============================
  // イベント登録
  // ===============================
  canvas.addEventListener("mousedown", startDraw);
  canvas.addEventListener("mousemove", draw);
  canvas.addEventListener("mouseup", endDraw);
  canvas.addEventListener("mouseleave", endDraw);

  canvas.addEventListener("touchstart", startDraw, { passive: false });
  canvas.addEventListener("touchmove", draw, { passive: false });
  canvas.addEventListener("touchend", endDraw);
  canvas.addEventListener("touchcancel", endDraw);

</script>
