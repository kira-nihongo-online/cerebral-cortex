        return;
    }

        "https://script.google.com/macros/s/AKfycbynod0rBS95yXLkUIYFOeZiTFHTUd6UJu578YxSESQgIFsA2EPpe7bE0GXNVbLe0Sx4Dg/exec?mode=receipt&id=" + encodeURIComponent(id)
    );

    const data = await response.json();

    console.log(data);

    receiptData.date = data.date;
    receiptData.count = data.count;
    receiptData.course = data.course;
    receiptData.history = data.history;

    changeReceipt(data.course);

}

function changeReceipt(course){

    const img = document.getElementById("receipt");

    const studentName = document.getElementById("studentName");
    const receiptDate = document.getElementById("receiptDate");

    studentName.textContent = receiptData.name;
    receiptDate.textContent = receiptData.date;

    // ===== 10回コース =====
    if(course === 10){

        let count = receiptData.count;

        if(count < 1) count = 1;
        if(count > 10) count = 10;

        img.src = "receipt10/receipt10_" + count + ".png";

    }

    // ===== 20回コース =====
    if(course === 20){

        let count = receiptData.count;

        if(count < 1) count = 1;
        if(count > 20) count = 20;

        img.src = "receipt20/receipt20_" + count + ".png";

    }

    // ===== 30回コース =====
    if(course === 30){

        let count = receiptData.count;

        if(count < 1) count = 1;
        if(count > 30) count = 30;

        img.src = "receipt30/receipt30_" + count + ".png";

    }

    // ===== 文字位置 =====
    studentName.style.left = receiptPos[course].name.left + "px";
    studentName.style.top  = receiptPos[course].name.top + "px";

    receiptDate.style.left = receiptPos[course].date.left + "px";
    receiptDate.style.top  = receiptPos[course].date.top + "px";

    // ===== 文字サイズ =====
    studentName.style.fontSize =
        receiptPos[course].name.size + "px";

    receiptDate.style.fontSize =
        receiptPos[course].date.size + "px";

    showTestDates(course);

    if(!showTestDate){
        showHistoryDates(course);
    }

}

});

function showTestDates(course){

    // 古い日付を削除
    document.querySelectorAll(".stamp-date").forEach(el => el.remove());

    if(!showTestDate) return;

    const area = document.querySelector(".receipt-area");

    for(let i=0;i<datePos[course].length;i++){

        const div = document.createElement("div");

        div.className = "stamp-date";

        div.textContent = testDates[i];

        div.style.left = datePos[course][i].x + "px";
        div.style.top  = datePos[course][i].y + "px";

        div.style.width = stampDateStyle[course].width + "px";
        div.style.fontSize = stampDateStyle[course].size + "px";
        div.style.color = stampDateStyle[course].color;

        area.appendChild(div);

    }

}

// ===== 実データの日付表示 =====
function showHistoryDates(course){

    // 古い日付を削除
    document.querySelectorAll(".stamp-date").forEach(el => el.remove());

    const area = document.querySelector(".receipt-area");

    if(!receiptData.history) return;

    for(let i = 0; i < datePos[course].length; i++){

        const historyDate = receiptData.history[i];

        // 履歴がない回は表示しない
        if(!historyDate) continue;

        const div = document.createElement("div");

        div.className = "stamp-date";

        // yyyy/MM/dd → yy/MM/dd
        div.textContent = historyDate.substring(2);

        div.style.left = datePos[course][i].x + "px";
        div.style.top  = datePos[course][i].y + "px";

        div.style.width = stampDateStyle[course].width + "px";
        div.style.fontSize = stampDateStyle[course].size + "px";
        div.style.color = stampDateStyle[course].color;

        area.appendChild(div);
    }
}

// ===== PNG保存 =====
function savePNG(){

    const receiptArea = document.querySelector(".receipt-area");

    html2canvas(receiptArea).then(canvas => {

        const link = document.createElement("a");

        link.download = receiptData.name + "_領収書.png";

        link.href = canvas.toDataURL("image/png");

        link.click();

    });

}
